CAPSTONE PROJECT 02

Tech & Cricket Genius (Quiz Application)

Build a multi-screen Quiz Application featuring state machine flow: Start Screen → Question Card → Option Selection → Score Calculation → Result Certificate Screen.

🇮🇳 Practical Capstone Project
In this hands-on project, students build Tech & Cricket Genius Quiz. It demonstrates conditional rendering of screen states, array navigation by question index, score accumulation, and game reset.

1. Quiz Application State Machine

Screen State Flow:
[ START SCREEN ] --(Click Start)--> [ QUESTION CARD 1..N ] --(Last Question)--> [ RESULT SCREEN ]

2. Complete Runnable Code Implementation

import React, { useState } from 'react';

const quizData = [
  {
    id: 1,
    question: "Which hook is used to handle side-effects in React?",
    options: ["useState", "useEffect", "useContext", "useReducer"],
    answerIndex: 1
  },
  {
    id: 2,
    question: "What is the primary role of ReactDOM?",
    options: ["Compile JSX", "Render Virtual DOM to real DOM", "Handle CSS animations", "Manage Redux store"],
    answerIndex: 1
  },
  {
    id: 3,
    question: "How do you pass data down from parent to child component?",
    options: ["Using State", "Using Redux", "Using Props", "Using HTML attributes"],
    answerIndex: 2
  }
];

export default function TechCricketQuizApp() {
  const [gameState, setGameState] = useState('START'); // 'START' | 'QUIZ' | 'RESULT'
  const [currentIdx, setCurrentIdx] = useState(0);
  const [score, setScore] = useState(0);

  const handleStart = () => {
    setScore(0);
    setCurrentIdx(0);
    setGameState('QUIZ');
  };

  const handleSelectOption = (selectedIdx) => {
    if (selectedIdx === quizData[currentIdx].answerIndex) {
      setScore(prev => prev + 1);
    }

    if (currentIdx + 1 < quizData.length) {
      setCurrentIdx(prev => prev + 1);
    } else {
      setGameState('RESULT');
    }
  };

  return (
    <div style={{ maxWidth: '500px', margin: '2rem auto', padding: '1.5rem', background: '#fff', border: '1px solid #e2e8f0', borderRadius: '12px' }}>
      {/* 1. START SCREEN */}
      {gameState === 'START' && (
        <div style={{ textAlign: 'center' }}>
          <h2>🏏 Tech & Cricket Genius Quiz</h2>
          <p>Test your ReactJS mastery with 3 rapid questions.</p>
          <button onClick={handleStart} style={{ padding: '10px 20px', cursor: 'pointer' }}>🚀 Start Quiz</button>
        </div>
      )}

      {/* 2. QUIZ QUESTION SCREEN */}
      {gameState === 'QUIZ' && (
        <div>
          <span style={{ fontSize: '0.8rem', color: '#64748b' }}>Question {currentIdx + 1} of {quizData.length}</span>
          <h3 style={{ margin: '0.5rem 0 1rem' }}>{quizData[currentIdx].question}</h3>
          
          <div style={{ display: 'flex', flexDirection: 'column', gap: '8px' }}>
            {quizData[currentIdx].options.map((option, idx) => (
              <button 
                key={idx} 
                onClick={() => handleSelectOption(idx)}
                style={{ padding: '10px', textAlign: 'left', cursor: 'pointer' }}
              >
                {idx + 1}. {option}
              </button>
            ))}
          </div>
        </div>
      )}

      {/* 3. RESULT SCREEN */}
      {gameState === 'RESULT' && (
        <div style={{ textAlign: 'center' }}>
          <h2>🎉 Quiz Completed!</h2>
          <h1>Your Score: {score} / {quizData.length}</h1>
          <p>{score === quizData.length ? "🏆 Perfect Score! You are a React Genius!" : "👍 Good effort! Review the topics and try again."}</p>
          <button onClick={handleStart}>🔄 Retake Quiz</button>
        </div>
      )}
    </div>
  );
}
