CAPSTONE PROJECT 01

Chai & Code Task Master (To-Do App)

Build a production-grade componentized To-Do Application combining state, array manipulation, form input handling, conditional rendering, and task filtering.

🇮🇳 Practical Capstone Project
In this hands-on project, students build Chai & Code Task Master. You will componentize the application into: TaskInput, TaskList, TaskItem, and TaskFilter.

1. Component Hierarchy Architecture

  • AppContainer: Holds main array state (tasks) and active filter state (ALL, COMPLETED, PENDING).
  • TaskForm: Controlled input component to submit new task strings.
  • TaskList: Maps through tasks array and renders individual TaskItem components with unique key={task.id}.
  • TaskItem: Displays task status badge, completion toggle checkbox, and delete button.

2. Complete Runnable Code Implementation

import React, { useState } from 'react';

// Child 1: Task Form Input
function TaskForm({ onAddTask }) {
  const [taskText, setTaskText] = useState('');

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!taskText.trim()) return;
    onAddTask(taskText.trim());
    setTaskText('');
  };

  return (
    <form onSubmit={handleSubmit} style={{ display: 'flex', gap: '8px', marginBottom: '1rem' }}>
      <input 
        type="text" 
        value={taskText} 
        onChange={(e) => setTaskText(e.target.value)} 
        placeholder="Add new React task..." 
        style={{ flex: 1, padding: '8px 12px' }}
      />
      <button type="submit">+ Add Task</button>
    </form>
  );
}

// Child 2: Single Task Row Item
function TaskItem({ task, onToggle, onDelete }) {
  return (
    <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '8px', borderBottom: '1px solid #e2e8f0' }}>
      <span 
        onClick={() => onToggle(task.id)}
        style={{ cursor: 'pointer', textDecoration: task.isCompleted ? 'line-through' : 'none', color: task.isCompleted ? '#64748b' : '#0f172a' }}
      >
        {task.isCompleted ? '✅' : '⏳'} {task.title}
      </span>
      <button onClick={() => onDelete(task.id)} style={{ color: '#dc2626' }}>Delete</button>
    </div>
  );
}

// Parent Capstone Component
export default function ChaiTaskMasterApp() {
  const [tasks, setTasks] = useState([
    { id: 1, title: 'Master React JSX & Components', isCompleted: true },
    { id: 2, title: 'Complete To-Do Capstone Project', isCompleted: false }
  ]);

  const handleAddTask = (title) => {
    const newTask = { id: Date.now(), title, isCompleted: false };
    setTasks(prev => [newTask, ...prev]); // Immutably prepend task
  };

  const handleToggleTask = (id) => {
    setTasks(prev => prev.map(t => t.id === id ? { ...t, isCompleted: !t.isCompleted } : t));
  };

  const handleDeleteTask = (id) => {
    setTasks(prev => prev.filter(t => t.id !== id));
  };

  return (
    <div style={{ maxWidth: '500px', margin: '2rem auto', padding: '1.5rem', background: '#fff', border: '1px solid #e2e8f0', borderRadius: '12px' }}>
      <h2>☕ Chai & Code Task Master</h2>
      <TaskForm onAddTask={handleAddTask} />
      
      {tasks.length === 0 ? (
        <p style={{ color: '#64748b' }}>🎉 All tasks completed! Take a Chai break.</p>
      ) : (
        tasks.map(task => (
          <TaskItem 
            key={task.id} 
            task={task} 
            onToggle={handleToggleTask} 
            onDelete={handleDeleteTask} 
          />
        ))
      )}
    </div>
  );
}
