📜 1. Defining Student Schema & Model

A Schema defines the shape of documents inside a collection. A Model is a constructor compiled from the Schema that provides the interface to query, insert, update, and delete documents in MongoDB.

Student Model Definition (`Student.js`)

// Student.js - Mongoose Schema and Model Definition
const mongoose = require('mongoose');

const studentSchema = new mongoose.Schema({
  rollNo: {
    type: Number,
    required: [true, 'Roll number is required'],
    unique: true,
    index: true
  },
  name: {
    type: String,
    required: [true, 'Student name is required'],
    trim: true,
    minlength: [2, 'Name must be at least 2 characters long']
  },
  email: {
    type: String,
    required: [true, 'Email is required'],
    lowercase: true,
    trim: true
  },
  age: {
    type: Number,
    min: [15, 'Age must be at least 15'],
    max: [60, 'Age cannot exceed 60']
  },
  branch: {
    type: String,
    required: true,
    enum: ['CSE', 'ECE', 'EEE', 'IT', 'MECH', 'CIVIL']
  },
  cgpa: {
    type: Number,
    min: [0, 'CGPA cannot be negative'],
    max: [10, 'CGPA cannot exceed 10.0'],
    default: 0.0
  },
  courses: [{
    type: String
  }],
  isEnrolled: {
    type: Boolean,
    default: true
  }
}, {
  timestamps: true // Automatically adds createdAt and updatedAt fields
});

// Compile Schema into a Model
const Student = mongoose.model('Student', studentSchema);

module.exports = Student;

🔒 Validations

Enforces `required`, `min`, `max`, `minlength`, `enum`, and custom error messages automatically.

⚙️ Options & Defaults

`trim`, `lowercase`, `default`, and `timestamps: true` handle clean data formatting.

🔑 Index & Unique

`unique: true` on `rollNo` prevents duplicate student entries in the database.

➕ 2. Create (Insert) Operations

Mongoose provides multiple methods to insert new document records into MongoDB:

Method A: Using `save()` (Instance Method)

const Student = require('./Student');

// 1. Create a new document instance
async function createSingleStudent() {
  try {
    const student = new Student({
      rollNo: 101,
      name: "Rahul Sharma",
      email: "rahul.s@example.com",
      age: 20,
      branch: "CSE",
      cgpa: 8.75,
      courses: ["Data Structures", "DBMS", "Web Dev"]
    });

    // 2. Save document to MongoDB
    const savedStudent = await student.save();
    console.log("✅ Student Created Successfully:", savedStudent);
  } catch (error) {
    console.error("❌ Save Error:", error.message);
  }
}

Method B: Using `Student.create()` (Model Method)

// Create and save in a single operation
async function quickCreateStudent() {
  try {
    const newStudent = await Student.create({
      rollNo: 102,
      name: "Ananya Roy",
      email: "ananya@example.com",
      age: 21,
      branch: "ECE",
      cgpa: 9.10,
      courses: ["VLSI", "Digital Signal Processing"]
    });
    console.log("✅ Quick Created Student:", newStudent);
  } catch (error) {
    console.error("❌ Creation Error:", error.message);
  }
}

Method C: Bulk Insert with `Student.insertMany()`

async function insertMultipleStudents() {
  const studentsList = [
    { rollNo: 103, name: "Priya Patel", email: "priya@example.com", age: 20, branch: "CSE", cgpa: 8.4 },
    { rollNo: 104, name: "Vikas Reddy", email: "vikas@example.com", age: 22, branch: "IT", cgpa: 7.9 },
    { rollNo: 105, name: "Sneha Verma", email: "sneha@example.com", age: 19, branch: "EEE", cgpa: 9.3 }
  ];

  try {
    const docs = await Student.insertMany(studentsList);
    console.log(`✅ Bulk inserted ${docs.length} students into database.`);
  } catch (error) {
    console.error("❌ Bulk Insert Error:", error.message);
  }
}

🔍 3. Read (Query) Operations

Mongoose provides powerful query helper methods to retrieve student documents from MongoDB.

1. Find All Students & Filter by Condition

// Fetch all enrolled students
async function getAllStudents() {
  const students = await Student.find();
  console.log("All Students:", students);
}

// Find students with filter conditions (CSE branch and CGPA >= 8.0)
async function getTopCSEStudents() {
  const topStudents = await Student.find({
    branch: 'CSE',
    cgpa: { $gte: 8.0 }
  });
  console.log("Top CSE Students:", topStudents);
}

2. Find Single Document (`findOne` & `findById`)

// Find by Roll Number
async function getStudentByRollNo(rollNo) {
  const student = await Student.findOne({ rollNo: rollNo });
  if (!student) {
    console.log("No student found with Roll No:", rollNo);
  } else {
    console.log("Student Found:", student.name);
  }
}

// Find by MongoDB _id
async function getStudentById(id) {
  const student = await Student.findById(id);
  console.log("Found by ID:", student);
}

3. Projection, Sorting, and Pagination

async function getSortedStudentRoster() {
  const result = await Student.find({ isEnrolled: true })
    .select('name rollNo branch cgpa -_id') // Include name, rollNo, branch, cgpa; Exclude _id
    .sort({ cgpa: -1 })                     // Sort by CGPA descending
    .limit(5)                               // Limit to top 5 results
    .skip(0);                               // Skip offset for pagination

  console.log("Ranked Top 5 Students:", result);
}

✏️ 4. Update Operations

Mongoose offers methods to update existing student records safely.

1. `findByIdAndUpdate()` or `findOneAndUpdate()` (Recommended)

Set { new: true, runValidators: true } to get the updated document back and enforce schema validations during update! 
// Update CGPA and return updated document
async function updateStudentCGPA(rollNo, newCgpa) {
  try {
    const updatedStudent = await Student.findOneAndUpdate(
      { rollNo: rollNo },
      { $set: { cgpa: newCgpa } },
      { new: true, runValidators: true } // Returns updated doc & runs validations
    );

    console.log("✅ Updated Student:", updatedStudent);
  } catch (error) {
    console.error("❌ Update failed:", error.message);
  }
}

// Add a new course to student's course list ($push array operator)
async function addCourseToStudent(rollNo, courseName) {
  const student = await Student.findOneAndUpdate(
    { rollNo: rollNo },
    { $addToSet: { courses: courseName } }, // $addToSet prevents duplicate entries
    { new: true }
  );
  console.log("✅ Updated Courses:", student.courses);
}

2. Bulk Update (`updateMany`)

// Increment age of all CSE students by 1 year
async function incrementCSEAge() {
  const result = await Student.updateMany(
    { branch: 'CSE' },
    { $inc: { age: 1 } }
  );
  console.log(`✅ Modified ${result.modifiedCount} student records.`);
}

🗑️ 5. Delete Operations

Methods to delete single or multiple student records from the database.

1. Delete Single Document (`deleteOne` / `findByIdAndDelete`)

// Delete by Roll Number
async function deleteStudentByRoll(rollNo) {
  const result = await Student.deleteOne({ rollNo: rollNo });
  if (result.deletedCount > 0) {
    console.log(`✅ Student with Roll No ${rollNo} deleted successfully.`);
  } else {
    console.log(`⚠️ No student found with Roll No ${rollNo}.`);
  }
}

// Find and delete (returns deleted document)
async function removeStudent(id) {
  const deletedStudent = await Student.findByIdAndDelete(id);
  console.log("Deleted Record:", deletedStudent);
}

2. Bulk Delete (`deleteMany`)

// Remove all students who are no longer enrolled
async function cleanupUnenrolledStudents() {
  const result = await Student.deleteMany({ isEnrolled: false });
  console.log(`✅ Removed ${result.deletedCount} unenrolled students.`);
}
⚠️ Warning: Calling Student.deleteMany({}) without a query filter will wipe out all records in the collection!
